@startuml
title Activity Diagram - Evaluation Loop (Sub-Activity)

|EvaluationController|
start
:Enter Loop;

repeat
    :Request Next Question;
    
    |QuestionRepository|
    :Query Database;
    :Return Question Data;
    
    |EvaluationController|
    :Prepare AI Submission;
    
    ' Fork for concurrent processing/monitoring
    fork
        |AIModelEngine|
        :Process Question Text;
        :Generate AI Response;
    fork again
        |EvaluationController|
        :Start Timeout Timer;
    end fork
    
    |EvaluationController|
    :Forward Answer to Scorer;
    
    |ScoringEngine|
    :Grade AI Response;
    :Calculate Intermediate Score;
    
    ' Fork for parallel saving and logging
    fork
        |QuestionRepository|
        :Save Intermediate Result;
    fork again
        |ScoringEngine|
        :Log Grading Status to Trace Console;
    end fork
    
    |EvaluationController|
    :Evaluate Loop Conditions;

repeat while (Should Continue Loop?) is ([Max Questions Not Reached AND No Stop Signal]) not ([Max Questions Reached OR Stop Signal Received])

:Exit Loop;
stop
@enduml
