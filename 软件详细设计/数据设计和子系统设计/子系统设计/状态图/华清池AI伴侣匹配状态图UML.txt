title State Diagram - Huaqing Pool AI Companion Matching Task

[*] --> Idle : Create Matching Session

Idle --> QuestionnairePending : MatchRequest(UserID) / ProvideQuestionnaire(QuestionnaireData)

state QuestionnairePending {
    [*] --> WaitingForUserAnswers
    WaitingForUserAnswers --> AnswersSubmitted : SubmitAnswers(AnswersLocation)
    WaitingForUserAnswers --> Cancelled : Cancel() [User timeouts or cancels]
    AnswersSubmitted --> [*]
}

QuestionnairePending --> Profiling : [Answers received successfully]
QuestionnairePending --> Cancelled : [User aborted]

state Profiling {
    [*] --> AnalyzingAnswers : ProcessAnswers(Answers)
    AnalyzingAnswers --> ProfileBuilt : BuildUserProfile(Interests, Personality, Needs)
    ProfileBuilt --> [*]
}

Profiling --> CandidateRetrieval : [User profile created]
Profiling --> Aborted : [Profile creation failed / Invalid answers]

state CandidateRetrieval {
    [*] --> FetchingCandidates : GetAIPartnerInformation(UserProfileFilter)
    FetchingCandidates --> CandidatesReady : [Candidates returned from Database]
    FetchingCandidates --> MatchFailed : [Database error or 0 candidates]
    CandidatesReady --> [*]
    MatchFailed --> [*]
}

CandidateRetrieval --> Matching : [CandidateList > 0]
CandidateRetrieval --> Aborted : [No candidates available / System Error]

state Matching {
    [*] --> CalculatingScores : ExecuteMatchingAlgorithm(UserProfile, CandidateList)
    CalculatingScores --> ResultGenerated : CalculateCompatibilityScores()
    ResultGenerated --> [*]
}

Matching --> DeliveringResult : [MatchingResult generated]
Matching --> Aborted : [Algorithm failure]

DeliveringResult --> Completed : ReturnMatchingReport(MatchingResult) [Report sent to User]

Completed --> [*]
Cancelled --> [*]
Aborted --> [*]

@enduml
